function Card({
  title,
  children,
  className = "",
}: {
  title?: string;
  children: React.ReactNode;
  className?: string;
}) {
  return (
    <section
      className={`rounded-[8px] border border-[#e7e7e7] bg-white shadow-[0_1px_2px_rgba(0,0,0,0.02)] ${className}`}
    >
      {title ? (
        <div className="border-b border-[#efefef] px-5 py-4">
          <h3 className="text-[13px] font-semibold tracking-[0.01em] text-[#343434]">
            {title}
          </h3>
        </div>
      ) : null}
      {children}
    </section>
  );
}
